#+name: generate-inventory
#+begin_src emacs-lisp :results value table :exports results :colnames'("Repository" "Skill" "Description")
    (require 'seq)
    (let* ((root (file-name-directory
                  (or (buffer-file-name) load-file-name default-directory)))
           (catalog (with-temp-buffer
                      (insert-file-contents (expand-file-name "skills-catalog.json" root))
                      (json-parse-buffer :object-type 'alist :array-type 'list)))
           ;; Newlines and | would break the table; org handles the rest.
           (clean (lambda (s)
                    (replace-regexp-in-string
                     "[ \t\r\n]+" " "
                     (replace-regexp-in-string "|" "∣" (string-trim (or s "")))))))

      (seq-mapcat
       (lambda (repo)
         (mapcar (lambda (skill)
                   (list (symbol-name (car repo))
                         (format "=%s=" (symbol-name (car skill)))
                         (funcall clean (alist-get 'description (cdr skill)))))
                 (alist-get 'skills (cdr repo))))
       (alist-get 'repos catalog)))
#+end_src
